const express = require('express'); const fs = require('fs'); const path = require('path'); const app = express(); const PORT = 3000; const DB_FILE = path.join(__dirname, 'filmes.json'); // 👉 COLE SUA CHAVE DA API DO TMDB AQUI: const TMDB_API_KEY = 'SUA_CHAVE_API_AQUI'; app.use(express.json()); // Cria o arquivo de banco de dados se ele ainda não existir if (!fs.existsSync(DB_FILE)) { fs.writeFileSync(DB_FILE, JSON.stringify([], null, 2)); } // ========================================== // 1. ROTAS DA API (BACKEND) // ========================================== // Rota para listar os filmes já salvos app.get('/api/filmes', (req, res) => { const filmes = JSON.parse(fs.readFileSync(DB_FILE, 'utf-8')); res.json(filmes); }); // Rota para salvar um novo filme vindo do painel admin app.post('/api/admin/filmes', (req, res) => { const novoFilme = req.body; const filmes = JSON.parse(fs.readFileSync(DB_FILE, 'utf-8')); if (filmes.some(f => f.tmdb_id === novoFilme.tmdb_id)) { return res.status(400).json({ erro: 'Este filme já está no seu catálogo!' }); } filmes.push(novoFilme); fs.writeFileSync(DB_FILE, JSON.stringify(filmes, null, 2)); console.log(`✅ Filme adicionado: ${novoFilme.titulo}`); res.status(201).json({ mensagem: 'Filme salvo com sucesso!' }); }); // Rota para remover um filme do catálogo app.delete('/api/admin/filmes/:id', (req, res) => { let filmes = JSON.parse(fs.readFileSync(DB_FILE, 'utf-8')); filmes = filmes.filter(f => f.tmdb_id != req.params.id); fs.writeFileSync(DB_FILE, JSON.stringify(filmes, null, 2)); res.json({ mensagem: 'Filme removido!' }); }); // ========================================== // 2. INTERFACE DO PAINEL ADMIN (FRONTEND) // ========================================== app.get('/', (req, res) => { res.send(` Painel Admin - Catálogo de Filmes

🎬 Painel Admin: Importar Filmes (TMDB)

📂 Filmes Salvos no Seu Site (0)

`); }); app.listen(PORT, () => { console.log(\`🚀 Painel Admin funcionando em: http://localhost:\${PORT}\`); });
